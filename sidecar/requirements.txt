# The sidecar's direct dependencies, as floors. This is the list people edit; machines do not install
# it. Every installer (setup.cmd / scripts/setup.ps1, scripts/setup.sh, `make sidecar-setup`, CI)
# installs requirements.lock.txt, the exact versions resolved from this file. Floors alone let each
# machine resolve whatever was newest on the day it ran setup, so two laptops set up a month apart
# ran different ultralytics/torch/polars/fastapi - and only one of those sets had ever met the real
# camera. The lock holds the versions of the dev venv that has (RTX 4060 + StreamCam).
#
# After changing this file: `make sidecar-lock` (add UPGRADE=<package> to move one pin), then commit both.
# torch/torchvision are pinned there as PyPI's CPU builds; scripts/cuda_torch.py swaps in the same
# versions' CUDA builds when an NVIDIA driver is present.

fastapi>=0.139
uvicorn[standard]>=0.51
ultralytics>=8.4
opencv-python-headless>=5.0
numpy>=2.3
pydantic>=2.13
pytest>=9.1
httpx>=0.28
psutil>=6.0

# Read/written by sidecar/tools/train_model.py for the export's data.yaml. ultralytics
# already pulls it in, but declaring it here means the tools do not depend on a
# transitive one that could move.
pyyaml>=6

# The tracker's linear assignment. Ultralytics' `.track()` (BoT-SORT, every frame) imports it but
# does not depend on it, and pip-installs it at runtime instead - which fails in a uv venv (no
# pip) or offline, so capture died with "No module named 'lap'" on a fresh setup.
lap>=0.5.12

# Only needed to run a custom Roboflow-exported model as `native` (see
# docs/DETECTOR_BACKENDS.md §1a). The .onnx the local inference server caches
# is a stock Ultralytics export, so YOLO() loads it directly.
# CPU build: portable default. For NVIDIA, swap it for a CUDA-12 build of
# onnxruntime-gpu — see README "ONNX on the GPU". Do not install both; they
# share the `onnxruntime` import name, so having both breaks either and
# uninstalling one deletes the shared package directory.
onnxruntime>=1.20
